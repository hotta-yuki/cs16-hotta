> def kiritori2(data_sumple, left, right):
>     """
>     波形のピーク付近を切り出す。
> 
>     ピーク位置を中心に左 left 点・右 right 点を取り出す。
>     配列端でピークが近い場合はゼロパディングで補完する。
> 
>     Parameters
>     ----------
>     data_sumple : ndarray  切り出し対象の 1D 波形
>     left        : int      ピーク左側の点数
>     right       : int      ピーク右側の点数
> 
>     Returns
>     -------
>     ndarray  長さ (left + right) の切り出し波形
>     
>     テール波のまわりの波形を切り出すために, この関数を使う。
>     
>     """
>     L_target    = left + right                    # 目標の切り出し長
>     datamaxhere = np.nanargmax(data_sumple)       # ピーク位置のインデックス
>     datamaxstart = datamaxhere - left
>     datamaxend   = datamaxhere + right
>     cut_data = data_sumple[datamaxstart:datamaxend]
> 
>     if len(cut_data) != L_target:
>         # 配列端に近い場合は長さが足りないのでゼロで埋める
>         padding_needed = L_target - len(cut_data)
>         if padding_needed > 0:
>             return np.pad(cut_data, (0, padding_needed), mode='constant')
>         else:
>             return cut_data[:L_target]   # 万が一長すぎる場合は切り捨て
>     print(f"cut_data.shape: {cut_data.shape}")
>     return cut_data
>

> def make_fftdata(data, dt_sumple):
>     """
>     1D 波形に対してゼロパディング付き FFT を行い、振幅スペクトルと周波数軸を返す。
> 
>     fft_N 点になるようにゼロパディングし、rfft で計算する。
>     rfft を使うため返り値の長さは fft_N//2 + 1 点になる点に注意。
>     (sourse_2.py の fft とは配列長が異なる)
> 
>     Parameters
>     ----------
>     data      : ndarray  入力波形 (長さ ≤ fft_N)
>     dt_sumple : float    サンプリング間隔 [s]
> 
>     Returns
>     -------
>     magnitude : ndarray  振幅スペクトル (長さ fft_N//2 + 1)
>     freq      : ndarray  周波数軸 [Hz]  (長さ fft_N//2 + 1)
>     """
>     howmanyzero = fft_N - len(data)
>     # 前後均等にゼロを挿入して fft_N 点に揃える
>     data_fft = np.concatenate(
>         [np.zeros(int(howmanyzero / 2)), data, np.zeros(int(howmanyzero / 2))], 0)
>     yf   = np.fft.rfft(data_fft) / (fft_N / 2)      # 正規化した片側スペクトル
>     freq = np.fft.rfftfreq(fft_N, d=dt_sumple)       # 対応する周波数軸
>     return np.abs(yf), freq

> def kiritori2(data_sumple, left, right):
>     """
>     波形のピーク付近を切り出す。
> 
>     ピーク位置を中心に左 left 点・右 right 点を取り出す。
>     配列端でピークが近い場合はゼロパディングで補完する。
> 
>     Parameters
>     ----------
>     data_sumple : ndarray  切り出し対象の 1D 波形
>     left        : int      ピーク左側の点数
>     right       : int      ピーク右側の点数
> 
>     Returns
>     -------
>     ndarray  長さ (left + right) の切り出し波形
>     
>     テール波のまわりの波形を切り出すために, この関数を使う。
>     
>     """
>     L_target    = left + right                    # 目標の切り出し長
>     datamaxhere = np.nanargmax(data_sumple)       # ピーク位置のインデックス
>     datamaxstart = datamaxhere - left
>     datamaxend   = datamaxhere + right
>     cut_data = data_sumple[datamaxstart:datamaxend]
> 
>     if len(cut_data) != L_target:
>         # 配列端に近い場合は長さが足りないのでゼロで埋める
>         padding_needed = L_target - len(cut_data)
>         if padding_needed > 0:
>             return np.pad(cut_data, (0, padding_needed), mode='constant')
>         else:
>             return cut_data[:L_target]   # 万が一長すぎる場合は切り捨て
>     print(f"cut_data.shape: {cut_data.shape}")
>     return cut_data
>

> def make_fftdata(data, dt_sumple):
>     """
>     1D 波形に対してゼロパディング付き FFT を行い、振幅スペクトルと周波数軸を返す。
> 
>     fft_N 点になるようにゼロパディングし、rfft で計算する。
>     rfft を使うため返り値の長さは fft_N//2 + 1 点になる点に注意。
>     (sourse_2.py の fft とは配列長が異なる)
> 
>     Parameters
>     ----------
>     data      : ndarray  入力波形 (長さ ≤ fft_N)
>     dt_sumple : float    サンプリング間隔 [s]
> 
>     Returns
>     -------
>     magnitude : ndarray  振幅スペクトル (長さ fft_N//2 + 1)
>     freq      : ndarray  周波数軸 [Hz]  (長さ fft_N//2 + 1)
>     """
>     howmanyzero = fft_N - len(data)
>     # 前後均等にゼロを挿入して fft_N 点に揃える
>     data_fft = np.concatenate(
>         [np.zeros(int(howmanyzero / 2)), data, np.zeros(int(howmanyzero / 2))], 0)
>     yf   = np.fft.rfft(data_fft) / (fft_N / 2)      # 正規化した片側スペクトル
>     freq = np.fft.rfftfreq(fft_N, d=dt_sumple)       # 対応する周波数軸
>     return np.abs(yf), freq

> 1.  密度推定の概要とモチベーション
> 2.  単一ガウス分布の限界

> 
> ## スライド3：機械学習における密度推定の位置づけ
> *   機械学習の3つの柱：
>     1.  **回帰（Regression）:** 出力値の予測
>     2.  **次元削減（Dimensionality Reduction）:** 特徴抽出
>     3.  **密度推定（Density Estimation）:** データが従う確率分布の表現
> *   本資料では、3つ目の「密度推定」に焦点を当てる。
> 
> ## スライド4：密度推定の目的
> *   データをコンパクトに表現する。
> *   未知のデータポイントの出現確率を予測する。
> *   データの背後にある生成メカニズムをモデル化する。
> 
> ## スライド5：単一ガウス分布の限界
> *   ガウス分布は「単峰性（Unimodal）」である。
> *   **課題:** データが複数のクラスターを持つ場合、平均と分散だけでは表現しきれない。
> *   **図11.1** ガウス分布では意味のある表現ができない二次元データセット。
> *   `![](images4bc5cd5a-33c0-4b05-a934-f9c5478edef8-01_576_855_1864_653.jpg)`
> 
> ## スライド6：混合モデルの導入
> *   より表現力の高い分布の族が必要。
> *   「単純な分布を組み合わせる」というアイデア。
> *   複数の「コンポーネント（基底分布）」を重み付けして足し合わせる（凸結合）。
> 
> ## スライド7：混合モデルの数学的定義
> *   混合分布 $p(\boldsymbol{x})$：
>     $$ p(\boldsymbol{x})=\sum_{k=1}^{K} \pi_{k} p_{k}(\boldsymbol{x}) $$
> *   制約条件：
>     1.  各重みは0以上1以下： $0 \leqslant \pi_{k} \leqslant 1$
>     2.  重みの総和は1： $\sum_{k=1}^{K} \pi_{k}=1$
> 
> ## スライド8：ガウス混合モデル（GMM）の定義
> *   基底分布 $p_k(\boldsymbol{x})$ としてガウス分布 $\mathcal{N}$ を採用：
>     $$ p(\boldsymbol{x} \mid \boldsymbol{\theta})=\sum_{k=1}^{K} \pi_{k} \mathcal{N}\left(\boldsymbol{x} \mid \boldsymbol{\mu}_{k}, \boldsymbol{\Sigma}_{k}\right) $$
> *   パラメータ集合 $\boldsymbol{\theta}$：
>     $$ \boldsymbol{\theta}:=\left\{\pi_{k}, \boldsymbol{\mu}_{k}, \boldsymbol{\Sigma}_{k}: k=1, \ldots, K\right\} $$
> 
> ## スライド9：GMMの視覚的理解
> *   **図11.2**：個々の加重ガウス成分（破線）と、それらを合成した混合密度（黒実線）。
> *   `![](images4bc5cd5a-33c0-4b05-a934-f9c5478edef8-03_510_761_434_699.jpg)`
> *   多峰性（Multimodal）のデータを滑らかに近似できる。
> 
> ## スライド10：最尤推定（Maximum Likelihood）
> *   目標：与えられたデータセット $\mathcal{X}$ を生成する確率（尤度）を最大化する $\boldsymbol{\theta}$ を見つける。
> *   独立同一分布（i.i.d.）の仮定：
>     $$ p(\mathcal{X} \mid \boldsymbol{\theta})=\prod_{n=1}^{N} p\left(\boldsymbol{x}_{n} \mid \boldsymbol{\theta}\right) $$
> 
> ## スライド11：対数尤度関数の構成
> *   計算を容易にするため対数をとる：
>     $$ \mathcal{L} = \sum_{n=1}^{N} \log \sum_{k=1}^{K} \pi_{k} \mathcal{N}\left(\boldsymbol{x}_{n} \mid \boldsymbol{\mu}_{k}, \boldsymbol{\Sigma}_{k}\right) $$
> *   **問題:** $\log$ の中に $\sum$ が入っているため、各パラメータに対して微分して=0と置いても、一発で解（解析解）が出ない。
> 
> ## スライド12：最適化へのアプローチ
> *   解析解がない $\rightarrow$ 反復的なアルゴリズムが必要。
> *   「他のパラメータを固定したまま、一つずつ更新する」という戦略。
> *   これが **EM（Expectation-Maximization）アルゴリズム**。
> 
> ## スライド13：重要概念：負担率（Responsibilities）
> *   各データポイント $\boldsymbol{x}_n$ がどの混合成分 $k$ から生成されたか？
> *   「ソフトな割り当て」の確率 $r_{nk}$ を定義：
>     $$ r_{n k} := \frac{\pi_{k} \mathcal{N}\left(\boldsymbol{x}_{n} \mid \boldsymbol{\mu}_{k}, \boldsymbol{\Sigma}_{k}\right)}{\sum_{j=1}^{K} \pi_{j} \mathcal{N}\left(\boldsymbol{x}_{n} \mid \boldsymbol{\mu}_{j}, \boldsymbol{\Sigma}_{j}\right)} $$
> 
> ## スライド14：負担率の性質
> *   $r_{nk}$ は事後確率を表す。
> *   全クラスターに対する合計は1になる： $\sum_{k} r_{nk} = 1$。
> *   **例11.2** の責任行列：データごとに各クラスターへの帰属度が数値化される。
> 
> ## スライド15：クラスターごとの総負担率
> *   データセット全体でクラスター $k$ がどれだけのデータを受け持っているか：
>     $$ N_{k} := \sum_{n=1}^{N} r_{n k} $$
> *   この $N_k$ が後の更新式で重要な役割を果たす。
> 
> ## スライド16：平均 $\boldsymbol{\mu}_k$ の更新式（導出）
> *   対数尤度を $\boldsymbol{\mu}_k$ で偏微分し、=0 と置くことで得られる：
>     $$ \boldsymbol{\mu}_{k}^{\text {new }} = \frac{1}{N_{k}} \sum_{n=1}^{N} r_{n k} \boldsymbol{x}_{n} $$
> *   これは「各データの負担率による加重平均」である。
> 
> ## スライド17：平均更新の視覚化
> *   **図11.4**：各平均値が、責任の大きい（負担率の高い）データポイントの方向へ引き寄せられるイメージ。
> *   `![](images4bc5cd5a-33c0-4b05-a934-f9c5478edef8-08_154_282_902_1459.jpg)`
> 
> ## スライド18：共分散 $\boldsymbol{\Sigma}_k$ の更新式
> *   同様に微分から導出：
>     $$ \boldsymbol{\Sigma}_{k}^{\text {new }} = \frac{1}{N_{k}} \sum_{n=1}^{N} r_{n k}\left(\boldsymbol{x}_{n}-\boldsymbol{\mu}_{k}\right)\left(\boldsymbol{x}_{n}-\boldsymbol{\mu}_{k}\right)^{\top} $$
> *   データが平均からどれだけ離れているかを、負担率で加重して計算する。
> 
> ## スライド19：混合重み $\pi_k$ の更新式
> *   和が1という制約条件のもとで最適化（ラグランジュの未定乗数法）：
>     $$ \pi_{k}^{\text {new }} = \frac{N_{k}}{N} $$
> *   クラスター $k$ の総負担率と、全データ数 $N$ の比率。
> 
> ## スライド20：EMアルゴリズムのフローチャート
> 1.  **初期化:** $\boldsymbol{\mu}_k, \boldsymbol{\Sigma}_{k}, \pi_k$ を適当な値で決める。
> 2.  **Eステップ（Expectation）:** 現パラメータを用いて負担率 $r_{nk}$ を計算。
> 3.  **Mステップ（Maximization）:** $r_{nk}$ を用いてパラメータを再推定（更新）。
> 4.  **収束判定:** 対数尤度の変化が小さくなるまで繰り返す。
> 
> ## スライド21：EMアルゴリズムの挙動（1次元の例）

> ## スライド22：EMアルゴリズムの収束プロセス
> *   **図11.8**：反復ごとに「負の対数尤度」が減少していく様子。
> *   `![](images4bc5cd5a-33c0-4b05-a934-f9c5478edef8-14_479_1191_1966_237.jpg)`

> ## スライド25：潜在変数モデルとしての解釈
> *   なぜ負担率を「事後確率」と呼べるのか？
> *   **潜在変数 $\boldsymbol{z}$** の導入：どのクラスターからデータが出たかを示す隠れた変数。
